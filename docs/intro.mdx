<script>
const perguntas = [
{
    pergunta: "Qual é o símbolo do Oxigênio?",
    opcoes: ["O", "Ox", "Og", "X"],
    correta: "O"
},
{
    pergunta: "Número atômico do Hidrogênio?",
    opcoes: ["1", "2", "8", "10"],
    correta: "1"
},
{
    pergunta: "Símbolo do Ouro?",
    opcoes: ["Au", "Ag", "O", "Gd"],
    correta: "Au"
},
{
    pergunta: "Elemento Na?",
    opcoes: ["Nitrogênio", "Sódio", "Neônio", "Níquel"],
    correta: "Sódio"
}
];

let atual = 0;
let pontos = 0;

function mostrarTela(id) {
    document.querySelectorAll(".tela").forEach(t => t.classList.remove("ativa"));
    document.getElementById(id).classList.add("ativa");
}

function novaPergunta() {
    document.getElementById("resultado").innerText = "";

    if (atual >= perguntas.length) {
        document.getElementById("final").innerText = pontos;
        mostrarTela("vitoria");
        return;
    }

    let p = perguntas[atual];

    document.getElementById("pergunta").innerText = p.pergunta;

    let opcoesHTML = "";
    p.opcoes.forEach(op => {
        opcoesHTML += `<button class="opcao" onclick="responder('${op}')">${op}</button>`;
    });

    document.getElementById("opcoes").innerHTML = opcoesHTML;

    atual++;
}

function responder(resposta) {
    let p = perguntas[atual - 1];

    if (resposta === p.correta) {
        document.getElementById("resultado").innerText = "✅ Acertou!";
        pontos++;
        document.getElementById("pontos").innerText = pontos;
    } else {
        document.getElementById("resultado").innerText = "❌ Errou! " + p.correta;
    }
}

function salvarRanking() {
    let nome = prompt("Seu nome:");
    if (!nome) return;

    let ranking = JSON.parse(localStorage.getItem("ranking")) || [];
    ranking.push({ nome, pontos });

    ranking.sort((a, b) => b.pontos - a.pontos);

    localStorage.setItem("ranking", JSON.stringify(ranking));
    verRanking();
}

function verRanking() {
    let ranking = JSON.parse(localStorage.getItem("ranking")) || [];
    let lista = "";

    ranking.forEach(r => {
        lista += `<li>${r.nome} - ${r.pontos} pts</li>`;
    });

    document.getElementById("listaRanking").innerHTML = lista;
    mostrarTela("ranking");
}

function reiniciar() {
    atual = 0;
    pontos = 0;
    document.getElementById("pontos").innerText = 0;
    mostrarTela("jogo");
    novaPergunta();
}

// começa direto
novaPergunta();
</script>
